D — VERIFICACION INDEPENDIENTE (SIN IMPORTAR EL MODULO DE REPLAY)
================================================================

Comprueba con aritmetica propia, leyendo la captura en solo lectura:
  1) que las 4 ternas alertadas cumplen las condiciones congeladas
     (0 < Q1 < Q2 < Q3, D1 > D2 > D3 > 0, cierres monotonos en la direccion);
  2) que entrada/salida son la apertura de las velas en los minutos exactos
     (primer minuto tras el corte y +5 min), exigiendo instantaneas coherentes;
  3) que el retorno alineado recalculado coincide con el publicado.

Fecha: 2026-11-18

1. SCRIPT (ejecutado tal cual, sin importar scripts/replay_patron_ventana4.py)

# -*- coding: utf-8 -*-
import io
import json
import sqlite3

DB = r"C:\proyectos\binance-futures-data\exp-v2-ventana-4-20261116a\research.db"
KL = r"C:\proyectos\binance-futures-data\exp-v2-ventana-4-20261116a\consultas_klines.jsonl"
EP = r"C:\proyectos\binance-futures-data\exp-v2-ventana-4-20261116a-patron-v1\episodios_patron.jsonl"
MIN = 60_000

con = sqlite3.connect("file:%s?mode=ro" % DB.replace("\\", "/"), uri=True)
con.row_factory = sqlite3.Row

crudo = {}
for linea in io.open(KL, encoding="utf-8"):
    d = json.loads(linea)
    for fila in json.loads(d["cuerpo"]):
        crudo.setdefault((d["symbol"], int(fila[0])), []).append(
            tuple(float(x) for x in fila[1:5])
        )

ok = True
for linea in io.open(EP, encoding="utf-8"):
    e = json.loads(linea)
    sym, dirn = e["symbol"], e["direccion"]
    tercera = int(e["terna"].split("|")[1])
    print("== %s %s  terna=%s  corte=%s" % (sym, dirn, e["terna"], e["cut_iso"]))
    qs, ds, ps = [], [], []
    for inicio in (tercera - 2 * MIN, tercera - MIN, tercera):
        f = con.execute(
            "select buy_notional,sell_notional,price_close from research_flow_1m "
            "where symbol=? and bucket_start=?", (sym, inicio)).fetchone()
        b = con.execute(
            "select bid20_mean,ask20_mean from research_book_1m "
            "where symbol=? and bucket_start=?", (sym, inicio)).fetchone()
        F = f["buy_notional"] - f["sell_notional"]
        D = b["ask20_mean"] if dirn == "LONG" else b["bid20_mean"]
        q = (1 if dirn == "LONG" else -1) * F / D
        qs.append(q); ds.append(D); ps.append(f["price_close"])
        print("   ventana %d  F=%+.4f  D=%.4f  Q=%+.6f  cierre=%.8g"
              % (inicio, F, D, q, f["price_close"]))
    cond = (0 < qs[0] < qs[1] < qs[2] and ds[0] > ds[1] > ds[2] > 0
            and (ps[0] < ps[1] < ps[2] if dirn == "LONG" else ps[0] > ps[1] > ps[2]))
    print("   patron_congelado=%s" % cond)
    ok = ok and cond

    cut = e["cut_ms"]
    entrada = (cut // MIN + 1) * MIN
    salida = entrada + 5 * MIN
    el = crudo.get((sym, entrada), [])
    sl = crudo.get((sym, salida), [])
    print("   corte=%d  entrada_esperada=%d  salida_esperada=%d" % (cut, entrada, salida))
    print("   velas crudas entrada=%s" % (el,))
    print("   velas crudas salida =%s" % (sl,))
    if not el or not sl or len(set(el)) != 1 or len(set(sl)) != 1:
        print("   RESULTADO: no verificable (vela ausente o instantaneas contradictorias)")
        ok = False
        continue
    print("   instantaneas: entrada=%d identicas=%s  salida=%d identicas=%s"
          % (len(el), len(set(el)) == 1, len(sl), len(set(sl)) == 1))
    pe, ps_ = el[0][0], sl[0][0]
    signo = 1 if dirn == "LONG" else -1
    ret = signo * (ps_ / pe - 1) * 1e4
    print("   entrada=%.10g salida=%.10g retorno_recalculado=%.4f  publicado=%.4f  coincide=%s"
          % (pe, ps_, ret, e["retorno_bps"], abs(ret - e["retorno_bps"]) < 1e-6))
    ok = ok and abs(ret - e["retorno_bps"]) < 1e-6

print()
print("VERIFICACION_INDEPENDIENTE_OK=%s" % ok)
